-- Добавляем колонку для кода пакета ALTER TABLE orders ADD COLUMN IF NOT EXISTS package_code VARCHAR(50); -- Обновляем триггер для использования package_code CREATE OR REPLACE FUNCTION process_order_materials() RETURNS TRIGGER AS $$ BEGIN -- Списываем основной материал (если есть пакет) IF NEW.package_code IS NOT NULL AND NEW.material_quantity > 0 THEN PERFORM create_movement( NEW.package_code, NEW.order_date, 'EXPENSE', NEW.material_quantity, 'Пакет для заказа ' || NEW.order_number ); END IF; -- Списываем сам материал IF NEW.material_quantity > 0 THEN PERFORM create_movement( NEW.material_type, NEW.order_date, 'EXPENSE', NEW.material_quantity, 'Материал для заказа ' || NEW.order_number ); END IF; -- Шапки IF NEW.has_caps AND NEW.material_quantity > 0 THEN PERFORM create_movement( 'NALEPKING', NEW.order_date, 'EXPENSE', NEW.material_quantity, 'Шапки для заказа ' || NEW.order_number ); END IF; -- Неформалки IF NEW.has_non_formal AND NEW.material_quantity > 0 THEN PERFORM create_movement( 'NONFORMAL_CAPS', NEW.order_date, 'EXPENSE', NEW.material_quantity, 'Неформалки для заказа ' || NEW.order_number ); END IF; -- Скобы (если есть шапки или неформалки) IF (NEW.has_caps OR NEW.has_non_formal) AND NEW.material_quantity > 0 THEN PERFORM create_movement( 'STAPLES', NEW.order_date, 'EXPENSE', NEW.material_quantity * 2, 'Скобы для заказа ' || NEW.order_number ); END IF; -- Картон IF NEW.has_cardboard AND NEW.material_quantity > 0 THEN PERFORM create_cardboard_movement( NEW.format, NEW.material_quantity, NEW.order_date, NEW.order_number ); END IF; -- ASTP IF NEW.has_astp AND NEW.astp_quantity > 0 THEN PERFORM create_movement( 'ASTP', NEW.order_date, 'EXPENSE', NEW.astp_quantity, 'ASTP для заказа ' || NEW.order_number ); END IF; -- Кванты IF NEW.has_quant AND NEW.quant_quantity > 0 THEN PERFORM create_movement( 'QUANT_BAG', NEW.order_date, 'EXPENSE', NEW.quant_quantity, 'Кванты для заказа ' || NEW.order_number ); END IF; -- Коробки IF NEW.has_boxes AND NEW.boxes IS NOT NULL AND jsonb_array_length(NEW.boxes) > 0 THEN FOR box_item IN SELECT * FROM jsonb_to_recordset(NEW.boxes) AS x(type TEXT, quantity INTEGER) LOOP PERFORM create_movement( box_item.type, NEW.order_date, 'EXPENSE', box_item.quantity, 'Коробки для заказа ' || NEW.order_number ); END LOOP; END IF; RETURN NEW; END; $$ LANGUAGE plpgsql; -- Пересоздаем триггер DROP TRIGGER IF EXISTS trg_process_order ON orders; CREATE TRIGGER trg_process_order AFTER INSERT ON orders FOR EACH ROW EXECUTE FUNCTION process_order_materials();
