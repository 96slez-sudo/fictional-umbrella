-- Добавляем новые колонки ALTER TABLE orders ADD COLUMN IF NOT EXISTS no_package BOOLEAN DEFAULT FALSE; ALTER TABLE orders ADD COLUMN IF NOT EXISTS boxes JSONB DEFAULT '[]'::jsonb; -- Обновляем триггер для обработки новых полей CREATE OR REPLACE FUNCTION process_order_materials() RETURNS TRIGGER AS $$ BEGIN -- Если НЕ "без пакета" - списываем основной материал IF NOT NEW.no_package THEN PERFORM create_movement( NEW.material_type, NEW.order_date, 'EXPENSE', NEW.material_quantity, 'Заказ ' || NEW.order_number ); END IF; -- Шапки (если есть) IF NEW.has_caps AND NOT NEW.no_package THEN PERFORM create_movement( 'NALEPKING', NEW.order_date, 'EXPENSE', NEW.material_quantity, 'Шапки для заказа ' || NEW.order_number ); END IF; -- Неформалки (если есть) IF NEW.has_non_formal AND NOT NEW.no_package THEN PERFORM create_movement( 'NALEPKING', -- Используем тот же материал NEW.order_date, 'EXPENSE', NEW.material_quantity, 'Неформалки для заказа ' || NEW.order_number ); END IF; -- Скобы (если есть шапки или неформалки) IF (NEW.has_caps OR NEW.has_non_formal) AND NOT NEW.no_package THEN PERFORM create_movement( 'STAPLES', NEW.order_date, 'EXPENSE', NEW.material_quantity * 2, 'Скобы для заказа ' || NEW.order_number ); END IF; -- Картон (если есть) IF NEW.has_cardboard AND NOT NEW.no_package THEN PERFORM create_cardboard_movement( NEW.format, NEW.material_quantity, NEW.order_date, NEW.order_number ); END IF; -- ASTP IF NEW.has_astp AND NEW.astp_quantity > 0 THEN PERFORM create_movement( 'ASTP', NEW.order_date, 'EXPENSE', NEW.astp_quantity, 'ASTP для заказа ' || NEW.order_number ); END IF; -- Кванты IF NEW.has_quant AND NEW.quant_quantity > 0 THEN PERFORM create_movement( 'QUANT_BAG', NEW.order_date, 'EXPENSE', NEW.quant_quantity, 'Кванты для заказа ' || NEW.order_number ); END IF; -- Коробки IF NEW.has_boxes AND NEW.boxes IS NOT NULL AND jsonb_array_length(NEW.boxes) > 0 THEN FOR box_item IN SELECT * FROM jsonb_to_recordset(NEW.boxes) AS x(type TEXT, quantity INTEGER) LOOP PERFORM create_movement( box_item.type, NEW.order_date, 'EXPENSE', box_item.quantity, 'Коробки для заказа ' || NEW.order_number ); END LOOP; END IF; RETURN NEW; END; $$ LANGUAGE plpgsql; -- Пересоздаем триггер DROP TRIGGER IF EXISTS trg_process_order ON orders; CREATE TRIGGER trg_process_order AFTER INSERT ON orders FOR EACH ROW EXECUTE FUNCTION process_order_materials();
